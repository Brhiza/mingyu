 const pathSegments = url.pathname.split('/').filter(Boolean);

: engine.calculate(validated);

// 统一证据链封装（对接 T4 法理面板）

// 为每个占卜类型 + 每个 spread（塔罗15种）自动生成工具定义
const TAROT_SPREADS = ['single','three','love','career','decision','celtic','chakra','year','mindBodySpirit','horseshoe','five','relationship','nine','element','grandTableau'];

export const generateAllTools = () => {
 const tools = [];

 tools.push({
 name: 'bazi_calculate',
 description: '计算八字四柱与十神',
 inputSchema: { type: 'object', properties: { year: { type: 'number' }, month: { type: 'number' }, day: { type: 'number' }, timeIndex: { type: 'string' } } },
 handler: async (args) => { /* 调用 @temposoul/core/bazi */ return { content: [{ type: 'text', text: '计算结果' }] }; }
 });

// 2. 塔罗 15 种牌阵自动化生成
 TAROT_SPREADS.forEach(spread => {
 tools.push({
 name: `tarot_${spread}`,
 description: `塔罗牌阵: ${spread}`,
 inputSchema: { type: 'object', properties: { question: { type: 'string' } } },
 handler: async (args) => {
 const { drawTarot } = await import('@temposoul/core/divination/tarot');
 const result = drawTarot({ spread, question: args.question });
 return { content: [{ type: 'text', text: JSON.stringify(result) }] };
 }
 });
 });

// 3. 其余占卜（六爻/梅花/奇门等）动态注册
 Object.keys(ENGINE_MAP).filter(k => k !== 'tarot').forEach(type => {
 tools.push({
 name: `${type}_calculate`,
 description: `执行 ${type} 占卜`,
 inputSchema: { type: 'object', properties: { question: { type: 'string' } } },
 handler: async (args) => {
 const engine = await ENGINE_MAP[type]();
 const result = engine.calculate(args);
 return { content: [{ type: 'text', text: JSON.stringify(result) }] };
 }
 });
 });

// 4. 风水工具（八宅/玄空等）
 ['bazhai', 'xuankong', 'residential'].forEach(type => {
 tools.push({
 name: `fengshui_${type}`,
 description: `风水分析: ${type}`,
 inputSchema: { type: 'object', properties: { sittingDirection: { type: 'string' }, facingDirection: { type: 'string' } } },
 handler: async (args) => {
 const engine = await import(`@temposoul/core/fengshui/${type}`);
 return { content: [{ type: 'text', text: JSON.stringify(engine.calculate(args)) }] };
 }
 });
 });

return tools;
};

import { Server } from '@modelcontextprotocol/sdk/server/index.js';
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js';
import { generateAllTools } from './tools/factory.js';

const allTools = generateAllTools();
const server = new Server({ name: 'temposoul-mcp', version: '0.3.0' }, { capabilities: { tools: {} } });

server.setRequestHandler('tools/list', async () => ({ tools: allTools.map(t => ({ name: t.name, description: t.description, inputSchema: t.inputSchema })) }));
server.setRequestHandler('tools/call', async (req) => {
 const tool = allTools.find(t => t.name === req.params.name);
 if (!tool) throw new Error('Tool not found');
 return tool.handler(req.params.arguments);
});

const transport = new StdioServerTransport();
await server.connect(transport);

```ts
import { create } from "zustand";

export type Locale =
  | "en"
  | "es"
  | "ar"
  | "th"
  | "pt"
  | "zh-CN"
  | "ja";
export type Direction = "ltr" | "rtl";

const SUPPORTED_LOCALES: Locale[] = [
  "en",
  "es",
  "ar",
  "th",
  "pt",
  "zh-CN",
  "ja",
];
const STORAGE_KEY = "temposoul-locale";

function detectLocale(): Locale {
  // 1. Check localStorage
  if (typeof window !== "undefined") {
    const saved = localStorage.getItem(STORAGE_KEY) as Locale | null;
    if (saved && SUPPORTED_LOCALES.includes(saved)) {
      return saved;
    }
  }

  // 2. Check browser language
  if (typeof navigator !== "undefined") {
    const browserLang = navigator.language;

    // Exact match
    if (SUPPORTED_LOCALES.includes(browserLang as Locale)) {
      return browserLang as Locale;
    }

    // Prefix match
    const prefix = browserLang.split("-")[0] as string;
    const matched = SUPPORTED_LOCALES.find(
      (l) => l.split("-")[0] === prefix
    );
    if (matched) return matched;
  }

  // 3. Default
  return "en";
}

function applyLocaleToDOM(locale: Locale): void {
  if (typeof document === "undefined") return;
  const dir: Direction = locale === "ar" ? "rtl" : "ltr";
  document.documentElement.lang = locale;
  document.documentElement.dir = dir;
}

interface I18nState {
  locale: Locale;
  direction: Direction;
  setLocale: (locale: Locale) => void;
}

export const useI18nStore = create<I18nState>((set) => {
  const initialLocale = detectLocale();
  applyLocaleToDOM(initialLocale);

  return {
    locale: initialLocale,
    direction: initialLocale === "ar" ? "rtl" : "ltr",
    setLocale: (locale) => {
      localStorage.setItem(STORAGE_KEY, locale);
      applyLocaleToDOM(locale);
      set({
        locale,
        direction: locale === "ar" ? "rtl" : "ltr",
      });
    },
  };
});
```

```ts
async function fetchWithRetry(
  url: string,
  options: RequestInit,
  retries: number = 0,
  isGet: boolean = false
): Promise<Response> {
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), 15000);

  try {
    const response = await fetch(url, {
      ...options,
      signal: controller.signal,
    });

    if (!response.ok && isGet && retries < 2) {
      const delay = Math.pow(3, retries) * 1000; // 1s, 3s
      await new Promise((r) => setTimeout(r, delay));
      return fetchWithRetry(url, options, retries + 1, isGet);
    }

    return response;
  } catch (err) {
    clearTimeout(timeout);
    if (
      isGet &&
      retries < 2 &&
      (err as Error).name !== "AbortError"
    ) {
      const delay = Math.pow(3, retries) * 1000;
      await new Promise((r) => setTimeout(r, delay));
      return fetchWithRetry(url, options, retries + 1, isGet);
    }
    throw err;
  } finally {
    clearTimeout(timeout);
  }
}

export async function apiRequest<T>(
  endpoint: string,
  options: RequestInit = {}
): Promise<{ data: T; meta: { service: string; version: string } }> {
  const url = `${API_BASE}${endpoint}`;
  const isGet = !options.method || options.method === "GET";

  const headers: Record<string, string> = {
    "Content-Type": "application/json",
    ...(options.headers as Record<string, string>),
  };

  // Attach auth token if available
  const { sessionToken } = useAppStore.getState() as any;
  if (sessionToken) {
    headers["Authorization"] = `Bearer ${sessionToken}`;
  }

  const response = await fetchWithRetry(url, { ...options, headers }, 0, isGet);

  if (!response.ok) {
    throw new ApiError(
      "HTTP_ERROR",
      `HTTP ${response.status}: ${response.statusText}`,
      response.status
    );
  }

  const json = await response.json();
  const parsed = ApiResponseSchema.parse(json);

  // Update engine version from meta
  if (parsed.meta?.version) {
    useAppStore.getState().setEngineVersion(parsed.meta.version);
  }

  if (!parsed.ok) {
    throw new ApiError(parsed.error.code, parsed.error.message);
  }

  return {
    data: parsed.data as T,
    meta: parsed.meta,
  };
}

export async function apiPost<T>(
  endpoint: string,
  body: unknown
): Promise<{ data: T; meta: { service: string; version: string } }> {
  return apiRequest<T>(endpoint, {
    method: "POST",
    body: JSON.stringify(body),
  });
}

export async function apiGet<T>(
  endpoint: string
): Promise<{ data: T; meta: { service: string; version: string } }> {
  return apiRequest<T>(endpoint, { method: "GET" });
}
```

```tsx
import zhCN from "./locales/zh-CN.json";
import es from "./locales/es.json";
import ar from "./locales/ar.json";
import th from "./locales/th.json";
import pt from "./locales/pt.json";
import ja from "./locales/ja.json";

const localeMap: Record<Locale, Record<string, string>> = {
  en,
  "zh-CN": zhCN,
  es,
  ar,
  th,
  pt,
  ja,
};

interface I18nContextType {
  t: (key: string, params?: Record<string, string | number>) => string;
  locale: Locale;
  direction: "ltr" | "rtl";
}

const I18nContext = createContext<I18nContextType>({
  t: (key: string) => key,
  locale: "en",
  direction: "ltr",
});

export function useTranslation() {
  return useContext(I18nContext);
}

export function I18nProvider({ children }: { children: ReactNode }) {
  const { locale, direction } = useI18nStore();

  const t = useMemo(() => {
    return (
      key: string,
      params?: Record<string, string | number>
    ): string => {
      const messages = localeMap[locale] || localeMap.en;
      let text =
        messages[key] ||
        localeMap.en[key] ||
        localeMap["zh-CN"][key] ||
        `[MISSING: ${key}]`;

      if (params) {
        Object.entries(params).forEach(([k, v]) => {
          text = text.replace(new RegExp(`\\{${k}\\}`, "g"), String(v));
        });
      }

      return text;
    };
  }, [locale]);

  return (
    <I18nContext.Provider value={{ t, locale, direction }}>
      {children}
    </I18nContext.Provider>
  );
}
```

```ts
import { useState, useEffect } from "react";

export function useMediaQuery(query: string): boolean {
  const [matches, setMatches] = useState(() => {
    if (typeof window === "undefined") return false;
    return window.matchMedia(query).matches;
  });

  useEffect(() => {
    const mql = window.matchMedia(query);
    const handler = (e: MediaQueryListEvent) => setMatches(e.matches);
    mql.addEventListener("change", handler);
    return () => mql.removeEventListener("change", handler);
  }, [query]);

  return matches;
}

export function useIsMobile(): boolean {
  return useMediaQuery("(max-width: 767px)");
}

export function useIsTablet(): boolean {
  return useMediaQuery("(min-width: 768px) and (max-width: 1023px)");
}

export function useIsDesktop(): boolean {
  return useMediaQuery("(min-width: 1024px)");
}

export function usePrefersReducedMotion(): boolean {
  return useMediaQuery("(prefers-reduced-motion: reduce)");
}
```

```ts
import { useEffect } from "react";
import { useI18nStore } from "@/store/i18nStore";

interface SEOParams {
  title: string;
  description: string;
  canonical?: string;
}

const LOCALE_MAP: Record<string, string> = {
  en: "en_US",
  "zh-CN": "zh_CN",
  es: "es_ES",
  ar: "ar_SA",
  th: "th_TH",
  pt: "pt_BR",
  ja: "ja_JP",
};

export function useSEO({ title, description, canonical }: SEOParams) {
  const { locale } = useI18nStore();

  useEffect(() => {
    document.title = title;

    // Update meta description
    let descMeta = document.querySelector('meta[name="description"]');
    if (!descMeta) {
      descMeta = document.createElement("meta");
      descMeta.setAttribute("name", "description");
      document.head.appendChild(descMeta);
    }
    descMeta.setAttribute("content", description);

    // Update og:title
    let ogTitle = document.querySelector('meta[property="og:title"]');
    if (!ogTitle) {
      ogTitle = document.createElement("meta");
      ogTitle.setAttribute("property", "og:title");
      document.head.appendChild(ogTitle);
    }
    ogTitle.setAttribute("content", title);

    // Update og:description
    let ogDesc = document.querySelector('meta[property="og:description"]');
    if (!ogDesc) {
      ogDesc = document.createElement("meta");
      ogDesc.setAttribute("property", "og:description");
      document.head.appendChild(ogDesc);
    }
    ogDesc.setAttribute("content", description);

    // Update og:locale
    let ogLocale = document.querySelector('meta[property="og:locale"]');
    if (!ogLocale) {
      ogLocale = document.createElement("meta");
      ogLocale.setAttribute("property", "og:locale");
      document.head.appendChild(ogLocale);
    }
    ogLocale.setAttribute("content", LOCALE_MAP[locale] || "en_US");

    // Update canonical
    if (canonical) {
      let link = document.querySelector('link[rel="canonical"]');
      if (!link) {
        link = document.createElement("link");
        link.setAttribute("rel", "canonical");
        document.head.appendChild(link);
      }
      link.setAttribute("href", canonical);
    }
  }, [title, description, canonical, locale]);
}
```

```ts
export function registerServiceWorker() {
  if (!("serviceWorker" in navigator)) return;

  window.addEventListener("load", async () => {
    try {
      const registration = await navigator.serviceWorker.register("/sw.js", {
        scope: "/",
      });

      // Check for updates periodically
      setInterval(() => {
        registration.update();
      }, 60 * 60 * 1000); // Every hour

      registration.addEventListener("updatefound", () => {
        const newWorker = registration.installing;
        if (!newWorker) return;

        newWorker.addEventListener("statechange", () => {
          if (
            newWorker.state === "installed" &&
            navigator.serviceWorker.controller
          ) {
            // New version available
            window.dispatchEvent(new CustomEvent("pwa-update-available"));
          }
        });
      });
    } catch (err) {
      console.error("SW registration failed:", err);
    }
  });
}
```

```tsx
interface Props {
  children: React.ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
  isChunkError: boolean;
}

export class ErrorBoundary extends React.Component<Props, State> {
  state: State = { hasError: false, error: null, isChunkError: false };

  static getDerivedStateFromError(error: Error): State {
    const isChunkError =
      error.name === "ChunkLoadError" ||
      error.message.includes("Loading chunk") ||
      error.message.includes("Loading CSS chunk");

    return { hasError: true, error, isChunkError };
  }

  componentDidCatch(error: Error, info: React.ErrorInfo) {
    console.error("[ErrorBoundary]", error, info.componentStack);

    // Auto-reload on chunk load error
    if (
      error.name === "ChunkLoadError" ||
      error.message.includes("Loading chunk")
    ) {
      window.location.reload();
      return;
    }

    // Report to error tracking service
    const reportUrl = window.__TEMPOSOUL_CONFIG__?.ERROR_REPORT_URL;
    if (reportUrl) {
      fetch(reportUrl, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          message: error.message,
          stack: error.stack,
          componentStack: info.componentStack,
          timestamp: new Date().toISOString(),
          url: window.location.href,
        }),
      }).catch(() => {
        // Silently fail - error reporting should not break the app
      });
    }
  }

  handleRetry = () => {
```

```tsx
import { useEffect, useRef, useCallback } from "react";
import { usePrefersReducedMotion } from "@/hooks/useMediaQuery";

interface Star {
  x: number;
  y: number;
  size: number;
  opacity: number;
  twinkleSpeed: number;
  layer: number;
}

export function StarfieldBackground() {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const animRef = useRef<number>(0);
  const starsRef = useRef<Star[]>([]);
  const mouseRef = useRef({ x: 0, y: 0 });
  const reducedMotion = usePrefersReducedMotion();

  const getParticleCount = useCallback(() => {
    if (reducedMotion) return 0;
    const cores = navigator.hardwareConcurrency || 2;
    const isMobile = window.innerWidth < 768;
    if (!isMobile && cores >= 4) return 1000;
    if (cores >= 4) return 400;
    return 200;
  }, [reducedMotion]);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    const resize = () => {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
    };
    resize();
    window.addEventListener("resize", resize);

    const count = getParticleCount();

    if (count === 0) {
      // Static fallback for prefers-reduced-motion
      ctx.fillStyle = "#0a0e1a";
      ctx.fillRect(0, 0, canvas.width, canvas.height);
      for (let i = 0; i < 120; i++) {
        ctx.beginPath();
        ctx.arc(
          Math.random() * canvas.width,
          Math.random() * canvas.height,
          Math.random() * 1.5 + 0.3,
          0,
          Math.PI * 2
        );
        ctx.fillStyle = `rgba(255, 255, 255, ${Math.random() * 0.7 + 0.1})`;
        ctx.fill();
      }
      return () => window.removeEventListener("resize", resize);
    }

    starsRef.current = Array.from({ length: count }, () => ({
      x: Math.random() * canvas.width,
      y: Math.random() * canvas.height,
      size: Math.random() * 2 + 0.3,
      opacity: Math.random() * 0.8 + 0.2,
      twinkleSpeed: Math.random() * 0.02 + 0.005,
      layer: Math.floor(Math.random() * 3),
    }));

    const parallaxFactors = [0.02, 0.05, 0.1];

    const animate = (time: number) => {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      const mx = (mouseRef.current.x - canvas.width / 2) / canvas.width;
      const my = (mouseRef.current.y - canvas.height / 2) / canvas.height;

      for (const star of starsRef.current) {
        const pf = parallaxFactors[star.layer]!;
        const ox = mx * pf * canvas.width;
        const oy = my * pf * canvas.height;
        const x =
          ((star.x + ox) % canvas.width + canvas.width) % canvas.width;
        const y =
          ((star.y + oy) % canvas.height + canvas.height) % canvas.height;
        const twinkle = Math.sin(time * star.twinkleSpeed) * 0.3 + 0.7;

        ctx.beginPath();
        ctx.arc(x, y, star.size, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(255, 255, 255, ${star.opacity * twinkle})`;
        ctx.fill();
      }

      animRef.current = requestAnimationFrame(animate);
    };

    const handleMouse = (e: MouseEvent) => {
      mouseRef.current = { x: e.clientX, y: e.clientY };
    };
    window.addEventListener("mousemove", handleMouse);

    const startAnimation = () => {
      animRef.current = requestAnimationFrame(animate);
    };

    if ("requestIdleCallback" in window) {
      (window as any).requestIdleCallback(startAnimation);
    } else {
      setTimeout(startAnimation, 200);
    }

    return () => {
      window.removeEventListener("resize", resize);
      window.removeEventListener("mousemove", handleMouse);
      cancelAnimationFrame(animRef.current);
    };
  }, [getParticleCount]);

  return (
    <canvas
      ref={canvasRef}
      aria-hidden="true"
      style={{
        position: "fixed",
        inset: 0,
        width: "100%",
        height: "100%",
        zIndex: "var(--z-starfield)" as any,
        pointerEvents: "none",
      }}
    />
  );
}
```

// temposoul-core/src/types/evidence.ts
// 专家A·证据型输出统一类型 —— 全引擎唯一事实源
// 白皮书 §2.3 契约2：引擎只产出「结构化证据 + 可移植 AI 提示词」

/** 出处类型：古籍 / 星历 / 天文标准 / 推导 */
export type SourceType = 'classic' | 'ephemeris' | 'standard' | 'derivation';

/** 计算链单步 */
export interface CalculationStep {
 /** 步骤序号，从 1 开始，全链连续 */
 step: number;
 /** 步骤名，如「确定年柱」「均时差修正」 */
 name: string;
 /** 公式的文字化表达（面向 T1/T4 展示），如 "EoT = 9.87·sin(2B) − 7.53·cos(B) − 1.5·sin(B)" */
 formula?: string;
 /** 该步的输入快照 */
 input: Record<string, unknown>;
 /** 该步的输出快照 */
 output: Record<string, unknown>;
 /** 出处定位，如 "Meeus, Astronomical Algorithms, ch.28" */
 reference?: string;
}

/** 证据条目 */
export interface EvidenceItem {
 /** 本证据支撑的断言，如「日主为庚金」 */
 claim: string;
 /** 出处名称，如《渊海子平》/ Meeus ch.28 */
 source: string;
 sourceType: SourceType;
 /** 精确定位：卷/章/页/公式号 */
 locator?: string;
}

/** 解释边界 */
export interface LimitationItem {
 /** 边界作用域，如 "true-solar-time" / "dayi-direction" */
 scope: string;
 description: string;
}

/** 反证 / 例外条件 */
export interface CounterEvidenceItem {
 /** 触发反证的条件，如「出生时刻处于夏令时期间」 */
 condition: string;
 /** 例外下的替代结论/口径 */
 alternative: string;
}

/** 证据容器：所有体系 calculate 结果必须 extends 此结构 */
export interface EvidenceEnvelope {
 calculationSteps: CalculationStep[];
 evidence: EvidenceItem[];
 limitation: LimitationItem[];
 counterEvidence: CounterEvidenceItem[];
}

/** 红线阈值常量（§4.1 分级表） */
export const EVIDENCE_GATE = {
 /** 警告级：反证最少条数 */
 MIN_COUNTER_EVIDENCE: 4,
 /** 警告级：limitation 最少条数 */
 MIN_LIMITATION: 1,
 /** 阻断级：calculationSteps 必须连续且从 1 开始 */
 STEPS_MUST_BE_CONTIGUOUS: true,
 /** 真太阳时对 Meeus 允差（秒）—— 阻断级 */
 TRUE_SOLAR_TIME_TOLERANCE_SECONDS: 1,
} as const;

/** 运行时守卫：任何体系结果组装出口处调用（§4.2 断言的函数化形态） */
export function assertEvidenceEnvelope(x: EvidenceEnvelope, systemId: string): void {
 const fail = (msg: string): never => {
 throw new Error(`[EVIDENCE_GATE][${systemId}]${msg}`);
 };
 if (!Array.isArray(x.calculationSteps) || x.calculationSteps.length === 0) {
 fail('calculationSteps 为空');
 }
 x.calculationSteps.forEach((s, i) => {
 if (s.step !== i + 1) fail(`calculationSteps 不连续：期望 step=${i + 1}，实际 step=${s.step}`);
 if (EVIDENCE_GATE.STEPS_MUST_BE_CONTIGUOUS && typeof s.name !== 'string') {
 fail(`step ${s.step} 缺少 name`);
 }
 });
 if (!Array.isArray(x.evidence) || x.evidence.length === 0) fail('evidence 为空');
 if (!Array.isArray(x.limitation) || x.limitation.length < EVIDENCE_GATE.MIN_LIMITATION) {
 fail(`limitation 少于 ${EVIDENCE_GATE.MIN_LIMITATION} 条`);
 }
 if (!Array.isArray(x.counterEvidence)) fail('counterEvidence 缺失');
}

/** 帮助函数：快速构建（减少各体系样板代码） */
export function makeEvidence(
 partial: Partial<EvidenceEnvelope> & { calculationSteps: CalculationStep[]; evidence: EvidenceItem[] },
): EvidenceEnvelope {
 return {
 limitation: partial.limitation ?? [],
 counterEvidence: partial.counterEvidence ?? [],
 ...partial,
 } as EvidenceEnvelope;
}

// 独立实现的 Meeus ch.28 均时差（作为对照基准，与引擎实现物理隔离，防止同源错误）
function meeusEotMinutes(dateUTC: Date): number {
 const rad = (d: number) => (d * Math.PI) / 180;
 const startOfYear = Date.UTC(dateUTC.getUTCFullYear(), 0, 1);
 const dayOfYear = (dateUTC.getTime() - startOfYear) / 86400000;
 const B = rad(360 * (dayOfYear - 81) / 364);
 return 9.87 * Math.sin(2 * B) - 7.53 * Math.cos(B) - 1.5 * Math.sin(B);
}

 if (!c.dateUTC) continue;
 const expected = meeusEotMinutes(new Date(c.dateUTC));
 const actual = calculateTrueSolarTime({
 dateUTC: new Date(c.dateUTC),
 longitude: 0,
 mode: 'eot-only',
 }).equationOfTimeMinutes;

// 主仓/scripts/quality-gates/run-gates.mjs
// 用法：node scripts/quality-gates/run-gates.mjs --config config/quality-gates/gates-A.json
// 输出：artifacts/quality-report.json；exitCode: 0=通过(警告不算)，1=存在 blocker 失败

import { execSync, spawnSync } from 'node:child_process';
import { readFileSync, writeFileSync, mkdirSync } from 'node:fs';

const argIdx = process.argv.indexOf('--config');
const configPath = argIdx === -1 ? 'config/quality-gates/gates-A.json' : process.argv[argIdx + 1];
const gates = JSON.parse(readFileSync(configPath, 'utf8')).gates;

const report = { blocked: [], warned: [], passed: [], timestamp: new Date().toISOString() };

// 按类型分组执行：test → vitest 一次跑全部（去重 target）
const testTargets = [...new Set(gates.filter((g) => g.type === 'test').map((g) => g.target))];
const scriptGates = gates.filter((g) => g.type === 'script');

for (const target of testTargets) {
 const res = spawnSync('npx', ['vitest', 'run', target, '--reporter=json'], { encoding: 'utf8' });
 const related = gates.filter((g) => g.target === target);
 const ok = res.status === 0;
 for (const g of related) {
 (ok ? report.passed : g.level === 'blocker' ? report.blocked : report.warned)
 .push({ id: g.id, target, detail: ok ? 'pass' : (res.stderr || res.stdout).slice(-500) });
 }
}

for (const g of scriptGates) {
 try {
 execSync(`node ${g.target}`, { stdio: 'pipe' });
 report.passed.push({ id: g.id, target: g.target, detail: 'pass' });
 } catch (e) {
 (g.level === 'blocker' ? report.blocked : report.warned)
 .push({ id: g.id, target: g.target, detail: String(e.stderr || e).slice(-500) });
 }
}

const exitCode = report.blocked.length > 0 ? 1 : 0;
mkdirSync('artifacts', { recursive: true });
writeFileSync('artifacts/quality-report.json', JSON.stringify(report, null, 2));

console.log(`\n[quality-gates] passed=${report.passed.length} warned=${report.warned.length} blocked=${report.blocked.length}`);
report.blocked.forEach((b) => console.error(` ✗ BLOCKER ${b.id}:${b.detail}`));
report.warned.forEach((w) => console.warn(` ⚠ WARNING ${w.id}`));
process.exit(exitCode);
